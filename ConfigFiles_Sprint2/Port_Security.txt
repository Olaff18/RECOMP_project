-- Munich

- SW7 (BR2 LAN: STAFF, ACCOUNTING, HR)

enable
configure terminal
ip dhcp snooping
ip dhcp snooping vlan 10,20,30
ip arp inspection vlan 10,20,30
no ip dhcp snooping information option
interface GigabitEthernet0/1
description Uplink to BR2 Router
switchport mode trunk
switchport nonegotiate
switchport trunk native vlan 50
switchport trunk allowed vlan 10,20,30,50
ip dhcp snooping trust
ip arp inspection trust
no shutdown
exit
interface range FastEthernet0/1 - 5
description ACCESS PORTS - STAFF
switchport mode access
switchport access vlan 10
switchport port-security
switchport port-security maximum 2
switchport port-security mac-address sticky
switchport port-security violation shutdown
spanning-tree portfast
spanning-tree bpduguard enable
no shutdown
exit
interface range FastEthernet0/6 - 10
description ACCESS PORTS - ACCOUNTING
switchport mode access
switchport access vlan 20
switchport port-security
switchport port-security maximum 2
switchport port-security mac-address sticky
switchport port-security violation shutdown
spanning-tree portfast
spanning-tree bpduguard enable
no shutdown
exit
interface range FastEthernet0/11 - 15
description ACCESS PORTS - HR
switchport mode access
switchport access vlan 30
switchport port-security
switchport port-security maximum 2
switchport port-security mac-address sticky
switchport port-security violation shutdown
spanning-tree portfast
spanning-tree bpduguard enable
no shutdown
exit
interface range FastEthernet0/16 - 24, GigabitEthernet0/2
description NOT IN USE - BLACKHOLE
switchport mode access
switchport access vlan 99
shutdown
exit
end
write memory
